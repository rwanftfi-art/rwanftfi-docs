  </Accordion>
  <Accordion title="مسؤولية المستخدم" icon="key">

  </Accordion>
  <Accordion title="حدود المسؤولية" icon="shield">

  </Accordion>
  <Accordion title="الترخيص الممنوح" icon="file-signature">
